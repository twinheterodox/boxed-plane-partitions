% Each example is symmetric in the floor coordinates. Transposition therefore
% gives the whole list of y-faces from the list of x-faces.
\newcommand{\StackFaces}[2]{%
  \foreach \i/\j/\k in {#1}{\TopTile{\i}{\j}{\k}}
  \foreach \i/\j/\k in {#2}{\XTile{\i}{\j}{\k}\YTile{\j}{\i}{\k}}}
\newcommand{\FlatStack}{%
  \StackFaces{0/0/1,1/0/1,2/0/1,0/1/1,1/1/1,0/2/1}%
             {3/0/0,2/1/0,1/2/0}}
\newcommand{\CubeStack}{%
  \StackFaces{0/0/2,1/0/1,0/1/1}{1/0/1,2/0/0,1/1/0}}

% The remaining visible squares of the floor and back walls complete the
% hexagon. In the flat example the stack already covers both back walls.
\newcommand{\FlatCorner}{\TopTile{2}{1}{0}\TopTile{1}{2}{0}\TopTile{2}{2}{0}}
\newcommand{\CubeCorner}{\TopTile{1}{1}{0}\XTile{0}{1}{1}\YTile{1}{0}{1}}

% Slice at the middle of each layer. The parts on exposed box walls continue
% the boundary of the occupied cells out to the two fixed endpoints.
\newcommand{\FlatSlice}{%
  \draw[slice,upperink] (0,3,.5)--(1,3,.5)--(1,2,.5)--(2,2,.5)--

  \fill[upperink] (3,0,.5) circle[radius=.69mm] node[right,endlabel] {\(A\)};}
\newcommand{\CubeSlices}{%
  \draw[slice,lowerink,dash pattern=on 2.5pt off 1.1pt]
    (0,2,.5)--(1,2,.5)--(1,1,.5)--(2,1,.5)--(2,0,.5);
  \draw[slice,upperink]
    (0,2,1.5)--(0,1,1.5)--(1,1,1.5)--(1,0,1.5)--(2,0,1.5);
  \fill[upperink] (0,2,1.5) circle[radius=.69mm] node[left,endlabel] {\(x\)};
  \fill[upperink] (2,0,1.5) circle[radius=.69mm] node[right,endlabel] {\(A\)};

  \fill[lowerink] (2,0,.5) circle[radius=.69mm] node[right,endlabel] {\(B\)};}

% The vertices of the path grid are midpoints of vertical unit edges.
% In these projected coordinates a step to the right adds (1,1) or (1,-1).
\newcommand{\PossibleSteps}[2]{%
  \begin{scope}[x={(.49cm,0cm)},y={(0cm,.2829cm)}]
    \clip ({-#1},{1-#1})--(0,{1-2*#1})--(#1,{1-#1})--
          (#1,{2*#2-#1-1})--(0,{2*#2-1})--({-#1},{2*#2-#1-1})--cycle;
    \pgfmathtruncatemacro{\lastLine}{#1+#2-1}
    \foreach \lineIndex in {0,...,\lastLine}{
      \pgfmathtruncatemacro{\intercept}{1-2*#1+2*\lineIndex}
      \draw[boxink!64,line width=.43pt,dash pattern=on .45pt off 1.43pt]
        ({-#1},{\intercept-#1})--(#1,{\intercept+#1})
        ({-#1},{\intercept+#1})--(#1,{\intercept-#1});}
  \end{scope}}

\newcommand{\LozengeView}[5]{% floor side, height, cube faces, bare faces, layer outlines
  \pgfmathsetmacro{\tileY}{.2829*(#1-#2)}
  \begin{scope}[shift={(1.55,\tileY)},x={(.49cm,-.2829cm)},
                y={(-.49cm,-.2829cm)},z={(0cm,.5658cm)}]
    #3
    \begin{scope}
      \colorlet{boxtop}{boxtop!20!white}
      \colorlet{boxleft}{boxleft!37!white}
      \colorlet{boxright}{boxright!37!white}
      #4
    \end{scope}
    \PossibleSteps{#1}{#2}
    \draw[boxink,line width=.78pt] (0,0,#2)--(#1,0,#2)--(#1,0,0)--
      (#1,#1,0)--(0,#1,0)--(0,#1,#2)--cycle;
    #5
  \end{scope}}
